use crate::errors::FrictionPointerError;
use chrono::{DateTime, Utc};
use serde::Deserialize;
use serde_json::Value;
use std::io::{self, BufRead};

/// Represents a single structured JSONL log record.
/// All fields use `#[serde(default)]` so standalone JSONL logs without an explicit
/// `line` number or `fields` map still parse cleanly; missing `line` values default
/// to `0` and are backfilled with the 1-based input line number during ingestion.
#[derive(Debug, Deserialize)]
pub struct Entry {
    #[serde(default)]
    pub line: usize,
    #[serde(default)]
    pub timestamp: Option<String>,
    #[serde(default)]
    pub level: Option<String>,
    #[serde(default)]
    pub message: Option<String>,
    #[serde(default)]
    pub fields: Option<serde_json::Map<String, Value>>,
}

impl Entry {
    pub fn parsed_timestamp(&self) -> Option<DateTime<Utc>> {
        self.timestamp
            .as_ref()
            .and_then(|s| s.parse::<DateTime<Utc>>().ok())
    }

    pub fn duration_ms(&self) -> Option<i64> {
        self.fields
            .as_ref()
            .and_then(|m| m.get("duration_ms"))
            .and_then(|v| v.as_i64())
    }
}

/// Reads structured JSONL entries from stdin into memory, aborting on the first
/// malformed JSON line so partial or corrupt inputs never produce misleading reports.
pub fn read_stdin() -> Result<Vec<Entry>, FrictionPointerError> {
    let stdin = io::stdin();
    let mut entries = Vec::new();

    // Enumerate before skipping blank lines so fallback 1-based line numbers stay aligned
    // with the original input stream.
    for (idx, line) in stdin.lock().lines().enumerate() {
        let line = line?;
        if line.trim().is_empty() {
            continue;
        }
        let mut entry: Entry = serde_json::from_str(&line)?;
        if entry.line == 0 {
            entry.line = idx + 1;
        }
        entries.push(entry);
    }

    Ok(entries)
}

use crate::reader::Entry;
use std::collections::HashMap;

/// Tracks 1-based line numbers for each exact non-empty message string
/// so repeated failures can be grouped without fuzzy matching or normalisation.
#[derive(Debug, Default)]
pub struct MessageCounts {
    counts: HashMap<String, Vec<usize>>,
}

impl MessageCounts {
    pub fn add(&mut self, entry: &Entry) {
        if let Some(msg) = &entry.message {
            self.counts
                .entry(msg.to_string())
                .or_default()
                .push(entry.line);
        }
    }

    pub fn repeated(&self, min_occurrences: usize) -> Vec<(String, Vec<usize>)> {
        self.counts
            .iter()
            .filter_map(|(msg, lines)| {
                if lines.len() >= min_occurrences {
                    Some((msg.clone(), lines.clone()))
                } else {
                    None
                }
            })
            .collect()
    }
}

use crate::counters::MessageCounts;
use crate::reader::Entry;
use chrono::{DateTime, Utc};

// Explicit, hardcoded thresholds keep friction detection deterministic across runs.
const SLOW_THRESHOLD_MS: i64 = 500;
const GAP_THRESHOLD_SECONDS: i64 = 60;
const REPEATED_MIN_OCCURRENCES: usize = 2;

#[derive(Debug)]
pub struct FrictionPoint {
    pub line: usize,
    pub timestamp: Option<String>,
    pub detail: String,
}

#[derive(Debug)]
pub struct SlowOp {
    pub line: usize,
    pub timestamp: Option<String>,
    pub duration_ms: i64,
    pub message: String,
}

#[derive(Debug)]
pub struct LongGap {
    pub from_line: usize,
    pub to_line: usize,
    pub from_ts: String,
    pub to_ts: String,
    pub gap_seconds: i64,
}

#[derive(Debug)]
pub struct RepeatedFailureGroup {
    pub message: String,
    pub occurrences: Vec<FrictionPoint>,
}

#[derive(Debug, Default)]
pub struct FrictionSummary {
    pub total_lines: usize,

    pub errors: Vec<FrictionPoint>,
    pub warnings: Vec<FrictionPoint>,
    pub slow_ops: Vec<SlowOp>,
    pub long_gaps: Vec<LongGap>,
    pub repeated_failures: Vec<RepeatedFailureGroup>,
}

/// Scans structured log entries in a single pass and groups explicit friction signals
/// without inferring severity or meaning from unstructured text.
pub fn analyse(entries: &[Entry]) -> FrictionSummary {
    let mut summary = FrictionSummary {
        total_lines: entries.len(),
        ..Default::default()
    };

    let mut prev_ts: Option<(DateTime<Utc>, usize)> = None;
    let mut msg_counts = MessageCounts::default();

    for e in entries {
        // Only exact uppercase "ERROR" and "WARN" levels are classified.
        if let Some(level) = &e.level {
            let point = FrictionPoint {
                line: e.line,
                timestamp: e.timestamp.clone(),
                detail: e.message.clone().unwrap_or_default(),
            };

            match level.as_str() {
                "ERROR" => summary.errors.push(point),
                "WARN" => summary.warnings.push(point),
                _ => {}
            }
        }

        // Slow operations rely strictly on numeric `fields.duration_ms`.
        if let Some(ms) = e.duration_ms() {
            if ms > SLOW_THRESHOLD_MS {
                summary.slow_ops.push(SlowOp {
                    line: e.line,
                    timestamp: e.timestamp.clone(),
                    duration_ms: ms,
                    message: e.message.clone().unwrap_or_default(),
                });
            }
        }

        // Only update `prev_ts` when a valid RFC3339 timestamp is present so
        // untimestamped entries do not break gap tracking between surrounding events.
        if let Some(ts) = e.parsed_timestamp() {
            if let Some((prev, prev_line)) = prev_ts {
                let gap = (ts - prev).num_seconds();
                if gap > GAP_THRESHOLD_SECONDS {
                    let from_ts = entries
                        .iter()
                        .find(|x| x.line == prev_line)
                        .and_then(|x| x.timestamp.clone())
                        .unwrap_or_else(|| prev.to_rfc3339());
                    let to_ts = e.timestamp.clone().unwrap_or_else(|| ts.to_rfc3339());

                    summary.long_gaps.push(LongGap {
                        from_line: prev_line,
                        to_line: e.line,
                        from_ts,
                        to_ts,
                        gap_seconds: gap,
                    });
                }
            }
            prev_ts = Some((ts, e.line));
        }

        msg_counts.add(e);
    }

    // Build repeated failure groups; `detail` is left empty on each occurrence because
    // the shared message text is stored once at the group level.
    for (msg, lines) in msg_counts.repeated(REPEATED_MIN_OCCURRENCES) {
        let mut occurrences = Vec::new();
        for line in lines {
            if let Some(entry) = entries.iter().find(|e| e.line == line) {
                occurrences.push(FrictionPoint {
                    line,
                    timestamp: entry.timestamp.clone(),
                    detail: String::new(),
                });
            }
        }
        summary.repeated_failures.push(RepeatedFailureGroup {
            message: msg,
            occurrences,
        });
    }

    summary
}

use crate::patterns::{FrictionSummary, LongGap, RepeatedFailureGroup};

/// Formats the analysed friction summary into a deterministic, human-readable report.
pub fn print(summary: &FrictionSummary) {
    println!("FrictionPointer Report");
    println!("======================\n");

    println!("Total:");
    println!("  Lines analysed: {}", summary.total_lines);
    println!("  Errors: {}", summary.errors.len());
    println!("  Warnings: {}", summary.warnings.len());
    println!("  Slow operations: {}", summary.slow_ops.len());
    println!("  Long gaps: {}", summary.long_gaps.len());
    println!("  Repeated failures: {}", summary.repeated_failures.len());
    println!();

    println!("Further Details:");
    println!("---------------\n");

    print_errors(summary);
    print_warnings(summary);
    print_slow_ops(summary);
    print_long_gaps(summary);
    print_repeated_failures(summary);
}

fn print_errors(summary: &FrictionSummary) {
    println!("Errors:");
    if summary.errors.is_empty() {
        println!("  (none)\n");
        return;
    }
    for e in &summary.errors {
        let ts = e.timestamp.as_deref().unwrap_or("-");
        println!("  [line {}]  {}  {}", e.line, ts, e.detail);
    }
    println!();
}

fn print_warnings(summary: &FrictionSummary) {
    println!("Warnings:");
    if summary.warnings.is_empty() {
        println!("  (none)\n");
        return;
    }
    for w in &summary.warnings {
        let ts = w.timestamp.as_deref().unwrap_or("-");
        println!("  [line {}]  {}  {}", w.line, ts, w.detail);
    }
    println!();
}

fn print_slow_ops(summary: &FrictionSummary) {
    println!("Slow Operations:");
    if summary.slow_ops.is_empty() {
        println!("  (none)\n");
        return;
    }
    for s in &summary.slow_ops {
        let ts = s.timestamp.as_deref().unwrap_or("-");
        println!("  [line {}]   [{} ms]   {}   {}", s.line, s.duration_ms, ts, s.message);
    }
    println!();
}

fn print_long_gaps(summary: &FrictionSummary) {
    println!("Long Gaps:");
    if summary.long_gaps.is_empty() {
        println!("  (none)\n");
        return;
    }
    for LongGap {
        from_line,
        to_line,
        from_ts,
        to_ts,
        gap_seconds,
    } in &summary.long_gaps
    {
        println!(
            "  [line {} → line {}]  {} → {} ({} seconds)",
            from_line, to_line, from_ts, to_ts, gap_seconds
        );
    }
    println!();
}

fn print_repeated_failures(summary: &FrictionSummary) {
    println!("Repeated Failures:");
    if summary.repeated_failures.is_empty() {
        println!("  (none)\n");
        return;
    }
    for RepeatedFailureGroup { message, occurrences } in &summary.repeated_failures {
        println!("  {} ({} occurrences)", message, occurrences.len());
        for o in occurrences {
            let ts = o.timestamp.as_deref().unwrap_or("-");
            println!("    - [line {}]  {}", o.line, ts);
        }
        println!();
    }
}

use crate::errors::FrictionPointerError;
use clap::{Parser, Subcommand};
use std::fs::File;

/// FrictionPointer — analyse structured JSONL logs for friction patterns.
///
/// This tool consumes JSON Lines (JSONL) logs produced by tools like LogTidier
/// and points out friction indicators such as:
///   • errors and warnings
///   • slow operations
///   • long gaps between events
///   • repeated failures
#[derive(Parser)]
#[command(
    name = "frictionpointer",
    version,
    about = "Analyse structured logs for friction patterns",
    long_about = None
)]
struct Cli {
    /// Kept optional so running `frictionpointer` with no subcommand prints a concise
    /// command overview rather than failing with a missing-argument error.
    #[command(subcommand)]
    command: Option<Commands>,
}

#[derive(Subcommand)]
enum Commands {
    /// Analyse structured JSONL logs (from file or stdin)
    Analyse {
        /// Optional input file (JSONL). If omitted, stdin is used.
        #[arg(short, long)]
        file: Option<PathBuf>,
    },

    /// Show example input/output
    Sample,

    /// Show condensed documentation
    Docs,

    /// Show tool information
    About,
}

fn main() -> Result<(), FrictionPointerError> {
    let cli = Cli::parse();

    match cli.command {
        Some(Commands::Analyse { file }) => {
            let entries = match file {
                Some(path) => read_file(&path)?,
                None => reader::read_stdin()?,
            };

            let summary = patterns::analyse(&entries);
            report::print(&summary);
        }

        Some(Commands::Sample) => {
            print_sample();
        }

        Some(Commands::Docs) => {
            print_docs();
        }

        Some(Commands::About) => {
            print_about();
        }

        None => {
            print_main_menu();
        }
    }

    Ok(())
}

/// Reads structured JSONL entries from disk, backfilling missing `line` fields with
/// their 1-based input line number and aborting on the first malformed JSON line.
fn read_file(path: &PathBuf) -> Result<Vec<reader::Entry>, FrictionPointerError> {
    let file = File::open(path)?;
    let reader = BufReader::new(file);

    let mut entries = Vec::new();
    for (idx, line) in reader.lines().enumerate() {
        let line = line?;
        if line.trim().is_empty() {
            continue;
        }
        let mut entry: reader::Entry = serde_json::from_str(&line)?;
        if entry.line == 0 {
            entry.line = idx + 1;
        }
        entries.push(entry);
    }

    Ok(entries)
}

# Documentation

## 1. Overview

FrictionPointer analyses **structured JSONL logs** and points out operational
friction patterns such as:

- errors and warnings  
- slow operations  
- long gaps between events  
- repeated failures  

It is intentionally strict, predictable, and designed to work standalone on
structured JSONL logs or compose cleanly with tools like LogTidier.  
FrictionPointer does **not** analyse unstructured raw logs or infer meaning from
free text messages.  
It relies entirely on explicit structured fields such as `timestamp`, `level`,
`message`, and `fields.duration_ms`.

If these fields are missing or invalid, the corresponding friction categories
cannot be detected.

FrictionPointer provides one primary mode:

- **Analyse**: read structured JSONL from file or stdin and emit a friction
  summary

---

## 2. Prerequisites

Before installing FrictionPointer, ensure the following requirements are met:

### 2.1 A terminal
FrictionPointer is a command‑line tool. You will need:

- **Windows**: PowerShell or Command Prompt (including VS Code's integrated
  terminal)  
- **macOS**: Terminal  
- **Linux**: Any shell (bash, zsh, etc.)

### 2.2 Understanding PATH
Your **PATH** is the list of folders your system searches when you type a
command.

If a program is inside a folder on your PATH, you can run it from any directory:

```

```

## 6. Input Format

FrictionPointer consumes **JSON Lines (JSONL)**.  
Each non-empty line must be a **valid JSON object**.  
A single malformed line aborts the entire run.

Supported fields:

- `line`: optional 1-based line number (defaults to the input line number if
  omitted)  
- `timestamp`: optional RFC3339 timestamp  
- `level`: optional log level (`"ERROR"` and `"WARN"` are reported)  
- `message`: optional log message  
- `fields`: optional JSON object containing structured metadata (such as
  numeric `duration_ms`)  

Example:

```json
{"line":1,"timestamp":"2024-01-01T12:00:00Z","level":"INFO","message":"Start","fields":{}}
{"line":2,"timestamp":"2024-01-01T12:00:01Z","level":"ERROR","message":"Failed","fields":{"duration_ms":800}}
{"line":3,"timestamp":"2024-01-01T12:00:02Z","level":"INFO","message":"Retry","fields":{}}
```

---

## 7. Behaviour

FrictionPointer applies strict, explicit rules.

### 7.1 JSON Parsing
- Every non-empty line must be valid JSON.  
- One malformed line aborts the entire analysis.  
- Empty or whitespace-only lines are skipped.

### 7.2 Level Detection
- Only `"ERROR"` and `"WARN"` (uppercase) are recognised.  
- `"error"`, `"Error"`, `"warning"` are ignored.  
- If `level` is missing, the event is not classified.

### 7.3 Slow Operation Detection
- Uses `fields.duration_ms` if present and numeric.  
- Slow operations are those greater than 500ms (hardcoded).  
- No inference from message text.

### 7.4 Long Gap Detection
- Requires a valid RFC3339 timestamp.  
- Gaps greater than 60 seconds between consecutive timestamped entries are
  reported (hardcoded).  
- Invalid or missing timestamps disable gap detection for that entry.

### 7.5 Repeated Failure Detection
- Non-empty `message` values are grouped by **exact string match**.  
- Case sensitive, whitespace sensitive, punctuation sensitive.  
- Minimum occurrences: 2 (hardcoded).  
- No fuzzy matching or normalisation.

### 7.6 Execution Model
- Entire log is loaded into memory.  
- No streaming mode.  
- No partial recovery.  
- Deterministic output.

---

## 8. Known Limitations

- Requires structured JSONL; raw unstructured logs are not supported  
- One malformed JSON line aborts the run  
- No streaming mode (entire file loaded into memory)  
- Only uppercase `"ERROR"` and `"WARN"` are detected  
- Only RFC3339 timestamps are supported  
- Thresholds are hardcoded (500ms slow ops, 60s gaps, 2 occurrences)  
- No fuzzy matching for repeated failures  
- No semantic grouping  
- No correlation analysis  
- No JSON output mode  
- Missing fields are silently ignored  

---

## 9. Troubleshooting

### Command not found
Ensure Cargo bin directory is on your PATH:

- **Windows**: `C:\Users\<you>\.cargo\bin\`
- **Linux/macOS**: `~/.cargo/bin/`

Restart your terminal after updating PATH.

### No friction detected
Check that your logs include:

- valid RFC3339 timestamps  
- uppercase `"ERROR"` or `"WARN"`  
- numeric `fields.duration_ms` values  

### JSON parse error
A single malformed line will abort the run.  
Validate your JSONL input.

FrictionPointer is a focused analysis tool for structured JSONL logs and
points out operational friction using explicit fields only, without inference or
heuristics. Its behaviour is deterministic, minimal, and designed to integrate
cleanly with real pipelines.
